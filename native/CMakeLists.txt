cmake_minimum_required(VERSION 3.24)
project(SyncEarAudio LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "SyncEar requires Windows Core Audio.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SyncEarAudio SHARED src/engine.cpp src/wasapi.cpp)
target_include_directories(SyncEarAudio PUBLIC include)
target_compile_definitions(SyncEarAudio PRIVATE SE_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SyncEarAudio PRIVATE /W4 /WX /wd4324 /permissive- /utf-8 /analyze)
target_link_libraries(SyncEarAudio PRIVATE ole32 uuid avrt)
include(CTest)
if(BUILD_TESTING)
  add_executable(audio_tests tests/dsp_tests.cpp)
  target_include_directories(audio_tests PRIVATE src)
  target_compile_options(audio_tests PRIVATE /W4 /WX /wd4324 /permissive-)
  add_test(NAME audio_dsp COMMAND audio_tests)
  add_executable(endpoint_smoke tests/endpoint_smoke.cpp)
  target_link_libraries(endpoint_smoke PRIVATE SyncEarAudio)
  target_compile_options(endpoint_smoke PRIVATE /W4 /WX /permissive-)
  add_test(NAME endpoint_enumeration COMMAND endpoint_smoke)
endif()
