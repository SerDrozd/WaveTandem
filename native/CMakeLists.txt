cmake_minimum_required(VERSION 3.24)
project(WaveTandemAudio LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "WaveTandem requires Windows Core Audio.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(WaveTandemAudio SHARED src/engine.cpp src/wasapi.cpp)
target_include_directories(WaveTandemAudio PUBLIC include)
target_compile_definitions(WaveTandemAudio PRIVATE SE_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(WaveTandemAudio PRIVATE /W4 /WX /permissive- /utf-8 /analyze)
target_link_libraries(WaveTandemAudio PRIVATE ole32 uuid avrt)
include(CTest)
if(BUILD_TESTING)
  add_executable(audio_tests tests/dsp_tests.cpp)
  target_include_directories(audio_tests PRIVATE src)
  target_compile_options(audio_tests PRIVATE /W4 /WX /permissive-)
  add_test(NAME audio_dsp COMMAND audio_tests)
  add_executable(endpoint_smoke tests/endpoint_smoke.cpp src/wasapi.cpp)
  target_include_directories(endpoint_smoke PRIVATE src)
  target_compile_definitions(endpoint_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(endpoint_smoke PRIVATE WaveTandemAudio ole32 uuid)
  target_compile_options(endpoint_smoke PRIVATE /W4 /WX /permissive-)
  add_test(NAME endpoint_enumeration COMMAND endpoint_smoke)
endif()
